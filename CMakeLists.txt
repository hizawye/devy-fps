cmake_minimum_required(VERSION 3.20)

project(devy_fps LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)

option(DEVY_ENABLE_IMGUI "Enable Dear ImGui debug UI" ON)
option(DEVY_WARNINGS_AS_ERRORS "Treat compiler warnings as errors" ON)
option(DEVY_ENABLE_CLANG_TIDY "Enable clang-tidy static analysis" OFF)

if(CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
  add_compile_options(-Wall -Wextra -Wpedantic -Wconversion -Wsign-conversion)
  if(DEVY_WARNINGS_AS_ERRORS)
    add_compile_options(-Werror)
  endif()
elseif(MSVC)
  add_compile_options(/W4 /permissive-)
  if(DEVY_WARNINGS_AS_ERRORS)
    add_compile_options(/WX)
  endif()
endif()

if(DEVY_ENABLE_CLANG_TIDY)
  find_program(CLANG_TIDY_BIN NAMES clang-tidy)
  if(CLANG_TIDY_BIN)
    set(CMAKE_CXX_CLANG_TIDY "${CLANG_TIDY_BIN}")
  else()
    message(WARNING "DEVY_ENABLE_CLANG_TIDY=ON but clang-tidy was not found.")
  endif()
endif()

# Dependencies (vcpkg manifest mode recommended)
find_package(SDL2 CONFIG REQUIRED)
find_package(OpenGL REQUIRED)
find_package(Bullet CONFIG REQUIRED)
find_package(glm CONFIG REQUIRED)
find_package(nlohmann_json CONFIG REQUIRED)
find_package(glad CONFIG REQUIRED)

find_package(enet CONFIG QUIET)
find_package(unofficial-enet CONFIG QUIET)
if(NOT enet_FOUND AND NOT TARGET unofficial::enet::enet)
  find_package(ENet CONFIG REQUIRED)
endif()

if(DEVY_ENABLE_IMGUI)
  find_package(imgui CONFIG REQUIRED)
endif()

add_subdirectory(shared)
add_subdirectory(engine)
add_subdirectory(client)
add_subdirectory(server)

include(CTest)
if(BUILD_TESTING)
  add_subdirectory(tests)
endif()
